# Schema SQL — 04A lifecycle (يدوي، بلا migration)

يُطبَّق على PostgreSQL **معزول** فقط. لا يسوّي ModelSnapshot للإنتاج.

## أعمدة SalesRequests الجديدة

```sql
ALTER TABLE "SalesRequests"
  ADD COLUMN IF NOT EXISTS "LinkedOrderId" integer NULL,
  ADD COLUMN IF NOT EXISTS "LastStatusNote" character varying(500) NULL,
  ADD COLUMN IF NOT EXISTS "ViewedAtUtc" timestamp with time zone NULL,
  ADD COLUMN IF NOT EXISTS "StatusChangedAtUtc" timestamp with time zone NULL,
  ADD COLUMN IF NOT EXISTS "ConcurrencyStamp" integer NOT NULL DEFAULT 0;

ALTER TABLE "SalesRequests"
  DROP CONSTRAINT IF EXISTS "FK_SalesRequests_Orders_LinkedOrderId";

ALTER TABLE "SalesRequests"
  ADD CONSTRAINT "FK_SalesRequests_Orders_LinkedOrderId"
  FOREIGN KEY ("LinkedOrderId") REFERENCES "Orders" ("Id") ON DELETE RESTRICT;

CREATE UNIQUE INDEX IF NOT EXISTS "IX_SalesRequests_LinkedOrderId"
  ON "SalesRequests" ("LinkedOrderId")
  WHERE "LinkedOrderId" IS NOT NULL;

CREATE INDEX IF NOT EXISTS "IX_SalesRequests_TargetEmployeeId"
  ON "SalesRequests" ("TargetEmployeeId");
```

## جدول Timeline

```sql
CREATE TABLE IF NOT EXISTS "SalesRequestTimelineEntries" (
  "Id" integer GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  "SalesRequestId" integer NOT NULL,
  "EventType" smallint NOT NULL,
  "ActorUserId" integer NOT NULL,
  "OccurredAtUtc" timestamp with time zone NOT NULL,
  "PreviousStatus" smallint NULL,
  "NewStatus" smallint NULL,
  "PreviousEmployeeId" integer NULL,
  "NewEmployeeId" integer NULL,
  "Reason" character varying(500) NULL,
  "OperationId" uuid NOT NULL,
  CONSTRAINT "FK_SalesRequestTimelineEntries_SalesRequests_SalesRequestId"
    FOREIGN KEY ("SalesRequestId") REFERENCES "SalesRequests" ("Id") ON DELETE CASCADE
);

CREATE INDEX IF NOT EXISTS "IX_SalesRequestTimelineEntries_SalesRequestId_OccurredAtUtc"
  ON "SalesRequestTimelineEntries" ("SalesRequestId", "OccurredAtUtc");
CREATE INDEX IF NOT EXISTS "IX_SalesRequestTimelineEntries_OperationId"
  ON "SalesRequestTimelineEntries" ("OperationId");
```
